-- CARROT GRAPHICS (OPTIMIZED FPS BOOST & DARK MODE)
--========================================================--
local fpsBoostActive = false

task.spawn(function()
    while true do
        if fpsBoostActive then
            pcall(function()
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
                Lighting.GlobalShadows = false
                Lighting.FogEnd = 5000 -- Ginawang mas maikli para sa dark/fog effect
                Lighting.Brightness = 0.2 -- Pinadilim para hindi masakit sa mata
                Lighting.ClockTime = 0 -- Ginawang gabi para sakto sa dark theme
                
                for _, v in ipairs(Lighting:GetChildren()) do
                    if v:IsA("PostEffect") or v:IsA("Atmosphere") or v:IsA("Sky") then
                        v:Destroy()
                    end
                end
                
                -- Batch processing para maiwasan ang pag-hang o lag spikes
                local count = 0
                for _, part in ipairs(Workspace:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.Material = Enum.Material.SmoothPlastic
                        part.Reflectance = 0
                    elseif part:IsA("Decal") or part:IsA("Texture") then
                        part.Transparency = 1
                    elseif part:IsA("ParticleEmitter") or part:IsA("Trail") or part:IsA("Fire") or part:IsA("Smoke") or part:IsA("Sparkles") then
                        part.Enabled = false
                    end
                    
                    count = count + 1
                    if count > 200 then
                        count = 0
                        task.wait() -- Nagbibigay ng breathing room sa engine para hindi mag-freeze
                    end
                end
            end)
        end
        task.wait(3) -- Pinalitan ng 3 seconds interval para hindi mabigat sa CPU
    end
end)

Workspace.DescendantAdded:Connect(function(v)
    if fpsBoostActive then
        task.spawn(function()
            pcall(function()
                if v:IsA("BasePart") then
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                elseif v:IsA("Decal") or v:IsA("Texture") then
                    v.Transparency = 1
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
                    v.Enabled = false
                end
            end)
        end)
    end
end)

    notifStroke.Color = Color3.fromRGB(0, 170, 255)
    notifStroke.Thickness = 1
    notifStroke.Parent = notifFrame

    local notifText = Instance.new("TextLabel")
    notifText.Size = UDim2.new(1, 0, 1, 0)
    notifText.BackgroundTransparency = 1
    notifText.Font = Enum.Font.GothamMedium
    notifText.Text = "  " .. message .. "  "
    notifText.TextColor3 = Color3.fromRGB(240, 240, 245)
    notifText.TextSize = 12
    notifText.ZIndex = 101
    notifText.Parent = notifFrame

    local textWidth = notifText.TextBounds.X + 20
    notifText.Size = UDim2.new(0, textWidth, 1, 0)
    notifFrame.Size = UDim2.new(0, textWidth, 0, 32)
    notifFrame.Position = UDim2.new(0.5, -textWidth / 2, 1, 50)

    local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local showTween = TweenService:Create(notifFrame, tweenInfo, {
        Position = UDim2.new(0.5, -textWidth / 2, 1, -45)
    })
    showTween:Play()

    task.spawn(function()
        task.wait(2)
        local hideTween = TweenService:Create(notifFrame, tweenInfo, {
            Position = UDim2.new(0.5, -textWidth / 2, 1, 50)
        })
        hideTween:Play()
        hideTween.Completed:Wait()
        notifFrame:Destroy()
    end)
end

local function openSelector(titleText, options, selectedTable, updateDisplayCallback)
    selectorTitle.Text = titleText
    for _, child in ipairs(selectorScroll:GetChildren()) do
        if child:IsA("TextButton") then
            child:Destroy()
        end
    end
    
    for i, opt in ipairs(options) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, -4, 0, 24)
        btn.BackgroundColor3 = selectedTable[opt] and Color3.fromRGB(0, 110, 180) or Color3.fromRGB(0, 55, 90)
        btn.BorderSizePixel = 0
        btn.Font = Enum.Font.GothamMedium
        btn.Text = "  " .. (selectedTable[opt] and "☑ " or "☐ ") .. opt
        btn.TextColor3 = Color3.fromRGB(220, 220, 230)
        btn.TextSize = 11
        btn.TextXAlignment = Enum.TextXAlignment.Left
        btn.ZIndex = 12
        btn.Parent = selectorScroll
        
        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 4)
        corner.Parent = btn
        
        btn.Activated:Connect(function()
            selectedTable[opt] = not selectedTable[opt]
            if selectedTable[opt] then
                btn.BackgroundColor3 = Color3.fromRGB(0, 110, 180)
                btn.Text = "  ☑ " .. opt
            else
                btn.BackgroundColor3 = Color3.fromRGB(0, 55, 90)
                btn.Text = "  ☐ " .. opt
            end
            updateDisplayCallback()
        end)
    end
    
    selectorScroll.CanvasSize = UDim2.new(0, 0, 0, #options * 27)
    selectorWindow.Visible = true
end

    button.Text = "Select >"
    button.TextColor3 = Color3.fromRGB(150, 150, 165)
    button.TextSize = 10
    button.TextXAlignment = Enum.TextXAlignment.Right
    button.AutoButtonColor = false
    button.Parent = parent

    local function updateText()
        local count = 0
        for _, v in pairs(selectedData) do
            if v then count = count + 1 end
        end
        if count == 0 then
            button.Text = "Select >"
        else
            button.Text = count .." selected >"
        end
    end

    button.Activated:Connect(function()
        openSelector("Select " .. text, options, selectedData, updateText)
    end)

    return button, selectedData
end

        if child:IsA("Frame") then

            if titleCheck and titleCheck.Text == "Run To Steal Timer" then
                local valCheck = child:GetChildren()
                for _, subChild in ipairs(valCheck) do
                    if subChild:IsA("TextLabel") and subChild ~= titleCheck then
                        subChild.Text = val .. "s"
                    end
                end
            end
        end
    end
end

            if indicator then indicator.Visible = true end

local function maskUsername(username)
    local length = #username
    if length == 1 then return "*" end
    if length == 2 then return username:sub(1,1) .. "*" end
    if length == 3 then return username:sub(1,1) .. "*" .. username:sub(3,3) end
    if length == 4 then return username:sub(1,1) .. "**" .. username:sub(4,4) end
    local visibleStart = 2
    local visibleEnd = 2
    local starCount = length - visibleStart - visibleEnd
    return username:sub(1,visibleStart) .. string.rep("*", starCount) .. username:sub(-visibleEnd)
end

    screenGui:Destroy()
end)

--========================================================--
-- LOGIC: STEADY ROTATION, CAMERA LOCKED TO GUARD, 2S WAIT
--========================================================--

task.spawn(function()
    while true do
        task.wait(1)
        
        local isToggleActive = false
        local isReturnToPlotActive = false
        
        if screenGui and screenGui.Parent then
            local mainFrameFound = screenGui:FindFirstChild("MainFrame")
            local contentAreaFound = mainFrameFound and mainFrameFound:FindFirstChild("ContentArea")
            local stealPanelFound = contentAreaFound and contentAreaFound:FindFirstChild("StealPanel")
            
            local speedCardFound = stealPanelFound and stealPanelFound:FindFirstChild("SpeedCard")
            local autoToggle = speedCardFound and speedCardFound:FindFirstChild("AutoStealToggle")
            
            if autoToggle then
                if autoToggle.BackgroundColor3.G > 0.4 or autoToggle.BackgroundColor3.R > 0.2 then
                    isToggleActive = true
                end
            end
            
            local returnCardFound = stealPanelFound and stealPanelFound:FindFirstChild("ReturnCard")
            local returnToggle = returnCardFound and returnCardFound:FindFirstChild("ReturnToPlot")
            
            if returnToggle then
                if returnToggle.BackgroundColor3.G > 0.4 or returnToggle.BackgroundColor3.R > 0.2 then
                    isReturnToPlotActive = true
                end
            end
        end
        
        if isToggleActive then
            local objectsFolder = Workspace:FindFirstChild("__OBJECTS")
            local areasFolder = objectsFolder and objectsFolder:FindFirstChild("Areas")
            local guardAreas = areasFolder and areasFolder:FindFirstChild("GuardAreas")
            
            if guardAreas then
                local char = player.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")
                local origHum = char and char:FindFirstChildOfClass("Humanoid")
                
                local currentSpeed = 16
                if screenGui then
                    local mainFrameFound = screenGui:FindFirstChild("MainFrame")
                    local contentAreaFound = mainFrameFound and mainFrameFound:FindFirstChild("ContentArea")
                    local stealPanelFound = contentAreaFound and contentAreaFound:FindFirstChild("StealPanel")
                    local speedCardFound = stealPanelFound and stealPanelFound:FindFirstChild("SpeedCard")
                    local speedInputBox = speedCardFound and speedCardFound:FindFirstChild("SpeedInput")
                    if speedInputBox then
                        currentSpeed = tonumber(string.match(speedInputBox.Text, "%d+")) or 16
                    end
                end
                if currentSpeed < 1 then currentSpeed = 16 end
                
                if hrp and origHum then
                    local cloneHum = origHum:Clone()
                    cloneHum.WalkSpeed    = currentSpeed
                    cloneHum.JumpPower    = origHum.JumpPower
                    cloneHum.MaxHealth    = origHum.MaxHealth
                    cloneHum.Health       = origHum.Health
                    cloneHum.AutoRotate   = false
                    cloneHum.Parent       = char
                    
                    workspace.CurrentCamera.CameraSubject = cloneHum
                    
                    task.wait(0.05)
                    if origHum and origHum.Parent then
                        origHum:Destroy()
                    end
                    
                    local forestParts = {}
                    local selectedParts = {}
                    local hasAnyAreaSelected = false
                    
                    for areaName, isSelected in pairs(globalSelectedAreas) do
                        if isSelected then
                            hasAnyAreaSelected = true
                            break
                        end
                    end
                    
                    local forestModelRef = nil
                    for _, areaModel in ipairs(guardAreas:GetChildren()) do
                        local areaName = areaModel.Name
                        
                        if areaName == "Forest" then
                            forestModelRef = areaModel
                            for _, part in ipairs(areaModel:GetDescendants()) do
                                if part:IsA("BasePart") then table.insert(forestParts, part) end
                            end
                            if areaModel:IsA("BasePart") then table.insert(forestParts, areaModel) end
                        end
                        
                        if globalSelectedAreas[areaName] then
                            for _, part in ipairs(areaModel:GetDescendants()) do
                                if part:IsA("BasePart") then table.insert(selectedParts, part) end
                            end
                            if areaModel:IsA("BasePart") then table.insert(selectedParts, areaModel) end
                        end
                    end
                    
                    if #forestParts > 0 then
                        local forestPart = forestParts[math.random(1, #forestParts)]
                        if char and char.PrimaryPart then
                            char:SetPrimaryPartCFrame(CFrame.new(forestPart.Position + Vector3.new(0, 3, 0)))
                        else
                            hrp.CFrame = CFrame.new(forestPart.Position + Vector3.new(0, 3, 0))
                        end
                        task.wait(0.1)
                        
                        for _, v in ipairs(workspace:GetDescendants()) do
                            if v:IsA("ProximityPrompt") and v.Name == "CarryAreaEgg" then
                                local parentPart = v.Parent
                                if parentPart and parentPart:IsA("BasePart") then
                                    local distance = (hrp.Position - parentPart.Position).Magnitude
                                    if distance <= 15 then
                                        pcall(function()
                                            v.HoldDuration = 0
                                            fireproximityprompt(v)
                                            v:InputHoldBegin()
                                            task.wait(0.05)
                                            v:InputHoldEnd()
                                        end)
                                    end
                                end
                            end
                        end
                        
                        task.wait(0.2)
                        if forestModelRef then
                            local guardPart = nil
                            for _, desc in ipairs(forestModelRef:GetDescendants()) do
                                if desc:IsA("BasePart") and (string.lower(desc.Name):find("guard") or string.lower(desc.Name):find("enemy") or desc.Name == "HumanoidRootPart") then
                                    guardPart = desc
                                    break
                                end
                            end
                            
                            if guardPart then
                                if char and char.PrimaryPart then
                                    char:SetPrimaryPartCFrame(guardPart.CFrame + Vector3.new(0, 1, 0))
                                else
                                    hrp.CFrame = guardPart.CFrame + Vector3.new(0, 1, 0)
                                end
                                workspace.CurrentCamera.CameraSubject = guardPart
                            end
                        end
                        
                        hrp.AssemblyLinearVelocity = Vector3.zero
                        hrp.AssemblyAngularVelocity = Vector3.zero
                        
                        local att = hrp:FindFirstChild("AntiFlingAttachment") or Instance.new("Attachment", hrp)
                        att.Name = "AntiFlingAttachment"
                        
                        local lv = hrp:FindFirstChild("AntiFlingForce") or Instance.new("LinearVelocity", hrp)
                        lv.Name = "AntiFlingForce"
                        lv.Attachment0 = att
                        lv.MaxForce = math.huge
                        lv.VectorVelocity = Vector3.zero
                        
                        local av = hrp:FindFirstChild("AntiFlingTorque") or Instance.new("AngularVelocity", hrp)
                        av.Name = "AntiFlingTorque"
                        av.Attachment0 = att
                        av.MaxTorque = math.huge
                        av.AngularVelocity = Vector3.zero
                        av.RelativeTo = Enum.ActuatorRelativeTo.World
                        
                        task.wait(1.5)
                    end
                    
                    if hasAnyAreaSelected and #selectedParts > 0 then
                        local att = hrp:FindFirstChild("AntiFlingAttachment")
                        local lv = hrp:FindFirstChild("AntiFlingForce")
                        local av = hrp:FindFirstChild("AntiFlingTorque")
                        if att then att:Destroy() end
                        if lv then lv:Destroy() end
                        if av then av:Destroy() end
                        
                        if char:FindFirstChildOfClass("Humanoid") then
                            workspace.CurrentCamera.CameraSubject = char:FindFirstChildOfClass("Humanoid")
                        end
                        
                        local selectedPart = selectedParts[math.random(1, #selectedParts)]
                        if char and char.PrimaryPart then
                            char:SetPrimaryPartCFrame(CFrame.new(selectedPart.Position + Vector3.new(0, 3, 0)))
                        else
                            hrp.CFrame = CFrame.new(selectedPart.Position + Vector3.new(0, 3, 0))
                        end
                        
                        task.wait(2.0)
                        
                        for _, v in ipairs(workspace:GetDescendants()) do
                            if v:IsA("ProximityPrompt") then
                                local pName = string.lower(v.Name)
                                local pAction = string.lower(v.ActionText or "")
                                if pName:find("egg") or pName:find("steal") or pAction:find("egg") or pAction:find("steal") or v.Name == "CarryAreaEgg" then
                                    local parentPart = v.Parent
                                    if parentPart and parentPart:IsA("BasePart") then
                                        local distance = (hrp.Position - parentPart.Position).Magnitude
                                        if distance <= 25 then
                                            pcall(function()
                                                v.HoldDuration = 0
                                                fireproximityprompt(v)
                                                v:InputHoldBegin()
                                                task.wait(0.05)
                                                v:InputHoldEnd()
                                            end)
                                        end
                                    end
                                end
                            end
                        end
                    end

                    if isReturnToPlotActive and hasAnyAreaSelected then
                        local currentHrp = char:FindFirstChild("HumanoidRootPart")
                        if currentHrp then
                            local returnDistance = (currentHrp.Position - SAFE_POS).Magnitude
                            local returnTime = returnDistance / currentSpeed
                            
                            local returnTweenInfo = TweenInfo.new(returnTime, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
                            local returnTween = TweenService:Create(currentHrp, returnTweenInfo, {CFrame = CFrame.new(SAFE_POS)})
                            returnTween:Play()
                            returnTween.Completed:Wait()
                            
                            currentHrp.AssemblyLinearVelocity = Vector3.zero
                            currentHrp.AssemblyAngularVelocity = Vector3.zero
                        end
                    end
                end
            end
        end
    end
end)
